import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="OpenAI-compatible" icon="approve-check">
    `/v1/chat/completions` and `/v1/models` speak the OpenAI wire verbatim — streaming or not, with tool calling, temperature + token-budget controls, and usage accounting. Existing SDKs point at higgs unchanged.
  </Card>
  <Card title="Crash-isolated worker" icon="warning">
    The llama.cpp engine runs in a separate process, spawned on load and reaped on unload — zero idle worker. A crash kills the worker, not your app; the supervisor restarts it and replays the last load.
  </Card>
  <Card title="Embeddable or standalone" icon="puzzle">
    Imports nothing from its host. Mount it into any Axum router with one call, or run the standalone `higgs` binary — same `/v1/*` + `/api/higgs/*` surface either way.
  </Card>
  <Card title="Finds your models" icon="list-format">
    Discovers GGUF files from LM Studio, HuggingFace cache, and Ollama layouts automatically — with just-in-time load, an idle reaper, and a pre-load RAM-fit guard.
  </Card>
  <Card title="Peer-to-peer remote fleet" icon="rocket">
    Pair other machines and route inference to their GPUs over an encrypted iroh QUIC link — no central server, no open ports. Durable routes, a live fleet view, and relayed remote-worker logs.
  </Card>
  <Card title="Pluggable engines" icon="setting">
    One `HiggsEngine` trait is the only seam to a backend. Add a new engine (e.g. MLX) by implementing the trait and adding one registry line; select it with `HIGGS_ENGINE`.
  </Card>
</CardGrid>
